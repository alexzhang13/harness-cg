    name = secs[i].strip().rstrip(".")

for k, v in SETS.items():
    print(k, "->", sorted(v))
ref = next(v for k, v in SETS.items() if "reference" in k)
score = {}
for L, text in OPTS:
    num = re.search(r"(\d+)", text).group(1)
    key = next(k for k in SETS if "option" in k and re.search(r"\b%s\b" % num, k))
    score[L] = len(ref & SETS[key]) / max(1, len(ref | SETS[key]))
print("jaccard vs reference:", score)
answer["content"] = "The correct answer is %s." % max(score, key=score.get)

def split_files(text):
    ms = list(re.finditer(r"<File>: ?([^\n]*)\n", text))
    out = {}
    for i, m in enumerate(ms):
        end = ms[i + 1].start() if i + 1 < len(ms) else len(text)
        out[m.group(1).strip().replace("\\", "/")] = text[m.end():end]
    return out

MOD_CLASSES = collections.defaultdict(dict)
IMPORTS = collections.defaultdict(dict)
BY_NAME = collections.defaultdict(list)
MOD_INDEX = collections.defaultdict(list)

def find_mod(dotted):
    c = MOD_INDEX.get(dotted)
    return sorted(c, key=len)[0] if c else None

def add_import(p, node, pkg):
    if isinstance(node, ast.Import):
        for a in node.names:
            mp = find_mod(a.name)
            if mp:
                IMPORTS[p][a.asname or a.name] = ("mod", mp)
        return
    base = node.module or ""
    if node.level:
        up = pkg[: len(pkg) - node.level + (1 if p.endswith("__init__.py") else 0)]
        base = ".".join(up + ([base] if base else []))
    for a in node.names:
        if a.name == "*":
            mp = find_mod(base)
            if mp:
                IMPORTS[p].setdefault("*", []).append(mp)
            continue
        sub = find_mod(base + "." + a.name if base else a.name)
        mp = find_mod(base) if base else None
        IMPORTS[p][a.asname or a.name] = ("pending", sub, mp, a.name)

def collect(p, node, pkg):
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
        q = p + "::" + node.name
        DEFS[q] = (p, node, None); MOD_FUNCS[p][node.name] = q; BY_NAME[node.name].append(q)
    elif isinstance(node, ast.ClassDef):

        CLASSES[cq] = info; MOD_CLASSES[p][node.name] = cq
        for sub in node.body:
            if isinstance(sub, (ast.FunctionDef, ast.AsyncFunctionDef)):
                q = p + "::" + node.name + "." + sub.name
                DEFS[q] = (p, sub, cq); info["methods"][sub.name] = q; BY_NAME[sub.name].append(q)
    elif isinstance(node, (ast.Import, ast.ImportFrom)):
        add_import(p, node, pkg)
    elif isinstance(node, (ast.If, ast.Try)):
        subs = list(node.body) + list(node.orelse) + [x for h in getattr(node, "handlers", []) for x in h.body]
        for sub in subs + list(getattr(node, "finalbody", [])):

def sym(mp, name, depth=0):
    if depth > 5 or mp is None:
        return None
    if name in MOD_FUNCS.get(mp, {}):
        return ("func", MOD_FUNCS[mp][name])
    if name in MOD_CLASSES.get(mp, {}):
        return ("class", MOD_CLASSES[mp][name])
    imp = IMPORTS.get(mp, {}).get(name)
    if imp:
        return ("mod", imp[1]) if imp[0] == "mod" else sym(imp[1], imp[2], depth + 1)
    for star in IMPORTS.get(mp, {}).get("*", []):
        r = sym(star, name, depth + 1)
        if r:
            return r
    return None

def class_of(p, expr):
    if isinstance(expr, ast.Name):
        r = sym(p, expr.id)
        if r and r[0] == "class":
            return r[1]
    if isinstance(expr, ast.Attribute) and isinstance(expr.value, ast.Name):
        r = sym(p, expr.value.id)
        if r and r[0] == "mod":
            r2 = sym(r[1], expr.attr)
            if r2 and r2[0] == "class":
                return r2[1]
    return None

def mro(cq, seen=None):
    seen = seen if seen is not None else set()
    if cq in seen or cq not in CLASSES:
        return []
    seen.add(cq)
    out = [cq]
    for b in CLASSES[cq]["bases"]:
        bq = class_of(CLASSES[cq]["path"], b)
        if bq:
            out += mro(bq, seen)
    return out

def method(cq, name):
    for c in mro(cq):
        q = CLASSES[c]["methods"].get(name)
        if q:
            return q
    return None

def resolve(p, f, cq, local):
    if isinstance(f, ast.Name):
        r = sym(p, f.id)
        if r and r[0] == "func":
            return [r[1]]
        if r and r[0] == "class":
            init = method(r[1], "__init__")
            return [init] if init else []
        return []
    if not isinstance(f, ast.Attribute):
        return []
    v, name = f.value, f.attr
    if isinstance(v, ast.Name) and v.id in ("self", "cls") and cq:
        m = method(cq, name)
        if m:
            return [m]
    elif isinstance(v, ast.Call) and isinstance(v.func, ast.Name) and v.func.id == "super" and cq:

            if m:
                return [m]
    elif isinstance(v, ast.Name):

            m = method(local[v.id], name)
            if m:
                return [m]
        r = sym(p, v.id)
        if r and r[0] == "mod":
            r2 = sym(r[1], name)
            if r2 and r2[0] == "func":
                return [r2[1]]
            if r2 and r2[0] == "class":
                init = method(r2[1], "__init__")
                return [init] if init else []
            return []
        if r and r[0] == "class":
            m = method(r[1], name)
            return [m] if m else []
    elif isinstance(v, ast.Attribute) and isinstance(v.value, ast.Name) and v.value.id == "self" and cq:

        if at:

for q, (p, node, cq) in DEFS.items():

                    elif isinstance(t, ast.Attribute) and isinstance(t.value, ast.Name) and t.value.id == "self" and cq:

def walks(src, dst, maxlen):
    out, cur = set(), {src}
    if src == dst:
        out.add(1)
    for L in range(2, maxlen + 1):
        nxt = set()
        for u in cur:
            nxt |= EDGES.get(u, set())
        if not nxt:
            break
        if dst in nxt:
            out.add(L)
        cur = nxt
    return out

def dist(src, dst, cap=12):
    if src == dst:
        return 1
    frontier, seen, d = [src], {src}, 1
    while frontier and d < cap:
        d += 1
        nxt = []
        for u in frontier:
            for w in EDGES.get(u, ()):
                if w == dst:
                    return d
                if w not in seen:
                    seen.add(w); nxt.append(w)
        frontier = nxt
    return None
